# The metrics push needs credentials for VictoriaMetrics. They come from the
# build environment (VM_USERNAME and VM_PASSWORD, passed with `docker run -e`)
# rather than from a committed file, and are written into a header inside the
# build directory, which is git-ignored. Neither value is ever printed. When
# either is empty the push is compiled out, so a checkout without them still
# builds and runs.
set(STAVVY_METRICS_USERNAME "$ENV{VM_USERNAME}")
set(STAVVY_METRICS_PASSWORD "$ENV{VM_PASSWORD}")

# A quote, a backslash or a line break in a password would break the generated
# string literal.
function(stavvy_escape_c_string value result)
    string(REPLACE "\\" "\\\\" escaped "${value}")
    string(REPLACE "\"" "\\\"" escaped "${escaped}")
    string(REPLACE "\n" "\\n" escaped "${escaped}")
    string(REPLACE "\r" "\\r" escaped "${escaped}")
    set(${result} "${escaped}" PARENT_SCOPE)
endfunction()

# metrics_credentials.h is generated here, after idf_component_register(), on
# purpose: ESP-IDF's early requirements scan includes this file with
# CMAKE_BUILD_EARLY_EXPANSION and returns from idf_component_register(), so
# nothing below that call runs during the scan and it cannot write a header
# outside the build directory.
set(STAVVY_METRICS_HEADER "")
string(APPEND STAVVY_METRICS_HEADER
    "// Generated by main/CMakeLists.txt from the VM_USERNAME and VM_PASSWORD\n"
    "// build environment variables. Never committed and never logged; do not edit.\n"
    "#pragma once\n\n"
)
if(STAVVY_METRICS_USERNAME STREQUAL "" OR STAVVY_METRICS_PASSWORD STREQUAL "")
    string(APPEND STAVVY_METRICS_HEADER "#define METRICS_PUSH_ENABLED 0\n")
else()
    stavvy_escape_c_string("${STAVVY_METRICS_USERNAME}" STAVVY_METRICS_ESCAPED_USERNAME)
    stavvy_escape_c_string("${STAVVY_METRICS_PASSWORD}" STAVVY_METRICS_ESCAPED_PASSWORD)
    string(APPEND STAVVY_METRICS_HEADER
        "#define METRICS_PUSH_ENABLED 1\n"
        "#define METRICS_USERNAME \"${STAVVY_METRICS_ESCAPED_USERNAME}\"\n"
        "#define METRICS_PASSWORD \"${STAVVY_METRICS_ESCAPED_PASSWORD}\"\n"
    )
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/metrics_credentials.h" "${STAVVY_METRICS_HEADER}")

target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
